CREATE TABLE `bitacora` (
  `id_bitacora` int(11) NOT NULL AUTO_INCREMENT,
  `id_usuar` int(11) NOT NULL,
  `id_sub` int(11) NOT NULL,
  `accion` varchar(50) NOT NULL,
  `detalle` text NOT NULL,
  `fecha` date NOT NULL,
  `hora` time NOT NULL,
  PRIMARY KEY (`id_bitacora`),
  KEY `id_usuar` (`id_usuar`),
  KEY `id_sub` (`id_sub`),
  CONSTRAINT `bitacora_ibfk_1` FOREIGN KEY (`id_usuar`) REFERENCES `usuarios` (`id_usuar`) ON DELETE CASCADE,
  CONSTRAINT `bitacora_ibfk_2` FOREIGN KEY (`id_sub`) REFERENCES `submodulos` (`id_sub`) ON DELETE CASCADE

-- 4. CREACIÓN DE VISTAS (VIEWS)
-- =========================================================
DROP VIEW IF EXISTS `v_personal_operativo_completo`;
CREATE VIEW `v_personal_operativo_completo` AS 
SELECT `po`.`id_po` AS `id_po`,`po`.`num_cobro` AS `num_cobro`,`po`.`nombres` AS `nombres`,
       `po`.`ape_pa` AS `ape_pa`,`po`.`ape_ma` AS `ape_ma`,`po`.`puesto` AS `puesto`,
       `po`.`estatus` AS `estatus`,`po`.`id_dep` AS `id_dep`,`d`.`nom_dep` AS `nom_dep`,
       (SELECT COUNT(`dsp`.`id_disp`) FROM `dispositivos` `dsp` WHERE `dsp`.`asig_po_id` = `po`.`id_po` OR `dsp`.`resp_po_id` = `po`.`id_po`) AS `total_equipos` 
FROM (`personal_operativo` `po` JOIN `departamentos` `d` ON(`po`.`id_dep` = `d`.`id_dep`));

DROP VIEW IF EXISTS `v_responsivas_personal`;
CREATE VIEW `v_responsivas_personal` AS 
SELECT `po`.`id_po` AS `id_po`,`po`.`num_cobro` AS `num_cobro`,
       TRIM(CONCAT(`po`.`nombres`,' ',`po`.`ape_pa`,' ',COALESCE(`po`.`ape_ma`,''))) AS `nombre_completo`,
       `po`.`estatus` AS `estatus`,
       (SELECT COUNT(`dsp`.`id_disp`) FROM `dispositivos` `dsp` WHERE `dsp`.`resp_po_id` = `po`.`id_po`) AS `total_equipos` 
FROM `personal_operativo` `po`;

-- 5. PROCEDIMIENTOS ALMACENADOS Y TRIGGERS (FORMATO SEGURO)
-- =========================================================
DELIMITER $$

-- PROCEDIMIENTO: sp_actualizar_ips_segmento
DROP PROCEDURE IF EXISTS `sp_actualizar_ips_segmento`$$
CREATE PROCEDURE `sp_actualizar_ips_segmento`(
    IN p_id_seg INT, 
    IN p_ip_ini VARCHAR(45), 
    IN p_ip_lim VARCHAR(45)
)
BEGIN
    DECLARE v_ip_actual INT UNSIGNED;
    DECLARE v_ip_final INT UNSIGNED;
    DECLARE v_ip_str VARCHAR(45);

    SET v_ip_actual = INET_ATON(p_ip_ini);
    SET v_ip_final = INET_ATON(p_ip_lim);

    -- A) Insertar nuevas IPs o RECICLAR las que estén inactivas
    WHILE v_ip_actual <= v_ip_final DO
        SET v_ip_str = INET_NTOA(v_ip_actual);
        
        INSERT INTO infraestructura_red (ip, est, id_seg) 
        VALUES (v_ip_str, 'Libre', p_id_seg)
        ON DUPLICATE KEY UPDATE 
            id_seg = IF(est = 'Inactivo', p_id_seg, id_seg),
            est = IF(est = 'Inactivo', 'Libre', est);
        
        SET v_ip_actual = v_ip_actual + 1;
    END WHILE;

    -- B) Eliminar IPs que quedaron fuera del rango (solo si están libres)
    DELETE FROM infraestructura_red 
    WHERE id_seg = p_id_seg 
      AND est = 'Libre' 
      AND (INET_ATON(ip) < INET_ATON(p_ip_ini) OR INET_ATON(ip) > INET_ATON(p_ip_lim));
END$$

-- PROCEDIMIENTO: sp_generar_ips_segmento
DROP PROCEDURE IF EXISTS `sp_generar_ips_segmento`$$
CREATE PROCEDURE `sp_generar_ips_segmento`(
    IN p_id_seg INT, 
    IN p_ip_ini VARCHAR(45), 
    IN p_ip_lim VARCHAR(45)
)
BEGIN
    DECLARE v_ip_actual INT UNSIGNED;
    DECLARE v_ip_final INT UNSIGNED;
    DECLARE v_ip_str VARCHAR(45);

    SET v_ip_actual = INET_ATON(p_ip_ini);
    SET v_ip_final = INET_ATON(p_ip_lim);

    WHILE v_ip_actual <= v_ip_final DO
        SET v_ip_str = INET_NTOA(v_ip_actual);
        
        -- Insertamos la IP. Si ya existía (por un borrado lógico previo), la "reciclamos"
        INSERT INTO infraestructura_red (ip, est, id_seg) 
        VALUES (v_ip_str, 'Libre', p_id_seg)
        ON DUPLICATE KEY UPDATE 
            id_seg = IF(est = 'Inactivo', p_id_seg, id_seg),
            est = IF(est = 'Inactivo', 'Libre', est);
            
        SET v_ip_actual = v_ip_actual + 1;
    END WHILE;
END$$

-- PROCEDIMIENTO: sp_sincronizar_personal_csv
DROP PROCEDURE IF EXISTS `sp_sincronizar_personal_csv`$$
CREATE PROCEDURE `sp_sincronizar_personal_csv`()
BEGIN
    -- =====================================================================
    -- FASE 1: RESOLUCIÓN ESTRICTA DE DEPARTAMENTOS
    -- =====================================================================
    
    -- Paso 1.1: Vincular los departamentos de la tabla temporal que YA existen en tu BD
    UPDATE temp_importacion_po t
    INNER JOIN departamentos d ON t.clave_dep_raw = d.clave_dep
    SET t.id_dep_final = d.id_dep
    WHERE t.id_dep_final IS NULL AND t.clave_dep_raw != '';

    -- Paso 1.2: Insertar en la tabla 'departamentos' los que aún están en NULL (son nuevos)
    INSERT INTO departamentos (clave_dep, nom_dep)
    SELECT DISTINCT clave_dep_raw, nom_dep_raw
    FROM temp_importacion_po
    WHERE id_dep_final IS NULL AND clave_dep_raw IS NOT NULL AND clave_dep_raw != '';

    -- Paso 1.3: Recuperar los IDs que se acaban de crear y pasarlos a la tabla temporal
    UPDATE temp_importacion_po t
    INNER JOIN departamentos d ON t.clave_dep_raw = d.clave_dep
    SET t.id_dep_final = d.id_dep
    WHERE t.id_dep_final IS NULL AND t.clave_dep_raw != '';

    -- =====================================================================
    -- FASE 2: INSERCIÓN EN PERSONAL_OPERATIVO (Con los IDs ya completos)
    -- =====================================================================

    -- Paso 2.1: Actualizar a los empleados que ya existían en el sistema
    UPDATE personal_operativo po
    INNER JOIN temp_importacion_po t ON po.num_cobro = t.num_cobro
    SET 
        po.nombres = t.nombres,
        po.ape_pa = t.ape_pa,
        po.ape_ma = t.ape_ma,
        po.puesto = t.puesto,
        po.correo = NULLIF(TRIM(t.correo), ''),
        po.dir = NULLIF(TRIM(t.dir), ''),         
        po.ext = NULLIF(TRIM(t.ext), ''),         
        po.cta_dom = NULLIF(TRIM(t.cta_dom), ''), 
        po.id_dep = COALESCE(t.id_dep_final, po.id_dep),
        po.estatus = 1;

    -- Paso 2.2: Insertar a los empleados que vienen en el CSV pero no en la BD
    INSERT INTO personal_operativo (num_cobro, nombres, ape_pa, ape_ma, puesto, correo, dir, ext, cta_dom, id_dep, estatus)
    SELECT 
        t.num_cobro, 
        t.nombres, 
        t.ape_pa, 
        t.ape_ma, 
        t.puesto, 
        NULLIF(TRIM(t.correo), ''), 
        NULLIF(TRIM(t.dir), ''),         
        NULLIF(TRIM(t.ext), ''),         
        NULLIF(TRIM(t.cta_dom), ''),     
        t.id_dep_final, 
        1
    FROM temp_importacion_po t
    LEFT JOIN personal_operativo po ON t.num_cobro = po.num_cobro
    WHERE po.id_po IS NULL;
    
END$$

-- TRIGGER: trg_red_sincronizar_estatus_bi
DROP TRIGGER IF EXISTS `trg_red_sincronizar_estatus_bi`$$
CREATE TRIGGER `trg_red_sincronizar_estatus_bi` BEFORE UPDATE ON `infraestructura_red` FOR EACH ROW 
BEGIN
    IF NEW.id_disp IS NOT NULL THEN
        SET NEW.est = 'Ocupado';
    ELSEIF NEW.id_disp IS NULL AND OLD.id_disp IS NOT NULL THEN
        IF OLD.est = 'Ocupado' THEN
            SET NEW.est = 'Libre';
        END IF;
    END IF;
END$$

-- TRIGGER: trg_generar_ips_after_insert_segmento
DROP TRIGGER IF EXISTS `trg_generar_ips_after_insert_segmento`$$
CREATE TRIGGER `trg_generar_ips_after_insert_segmento` AFTER INSERT ON `segmentos` FOR EACH ROW 
BEGIN
    CALL sp_generar_ips_segmento(NEW.id_seg, NEW.ip_ini, NEW.ip_lim);
END$$

-- TRIGGER: trg_actualizar_ips_after_update_segmento
DROP TRIGGER IF EXISTS `trg_actualizar_ips_after_update_segmento`$$
CREATE TRIGGER `trg_actualizar_ips_after_update_segmento` AFTER UPDATE ON `segmentos` FOR EACH ROW 
BEGIN
    IF OLD.ip_ini != NEW.ip_ini OR OLD.ip_lim != NEW.ip_lim THEN
        CALL sp_actualizar_ips_segmento(NEW.id_seg, NEW.ip_ini, NEW.ip_lim);
    END IF;
END$$
